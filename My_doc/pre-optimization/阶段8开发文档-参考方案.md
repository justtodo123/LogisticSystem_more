SYSTEM_PROMPT = """你是一个物流调度专家。根据用户需求和当前系统状态，生成算法参数JSON。

  }

如果用户需求不明确，使用默认参数（traditional算法，标准权重）。
"""

def build_user_prompt(order_count, vehicle_count, node_count, pending_orders):
    """构建用户提示词"""
    orders_desc = "\n".join([
        f"- 订单{order.order_code}: 目的地{node_name}, 时效{order.time_window}"
        for order in pending_orders[:10]  # 最多10个订单
    ])
    
    return f"""当前系统状态：
- 待分配订单：{order_count}个
- 可用车辆：{vehicle_count}辆
- 节点数量：{node_count}个

订单列表（前10个）：
{orders_desc}

用户需求：{user_message}

请生成算法参数JSON。
"""
```

### 1.4 API调用实现

```python
# 文件：src/backend/services/deepseek_service.py

import httpx
import json
from typing import Dict, Optional
from ...config import settings

class DeepSeekService:
    """DeepSeek API调用服务"""
    
    @staticmethod
    async def parse_natural_language(user_message: str, system_context: Dict) -> Dict:
        """
        解析自然语言，生成算法参数
        
        Args:
            user_message: 用户自然语言输入
            system_context: 系统上下文（订单数、车辆数等）
            
        Returns:
            {
                "success": bool,
                "algorithm_params": Dict,  # 成功时返回
                "raw_response": str,         # DeepSeek原始响应
                "error": str                 # 失败时返回
            }
        """
        try:
            # 1. 构建提示词
            user_prompt = build_user_prompt(**system_context)
            
            # 2. 调用DeepSeek API
            async with httpx.AsyncClient(timeout=30.0) as client:
                response = await client.post(
                    f"{settings.DEEPSEEK_API_BASE}/chat/completions",
                    headers={
                        "Authorization": f"Bearer {settings.DEEPSEEK_API_KEY}",
                        "Content-Type": "application/json"
                    },
                    json={
                        "model": "deepseek-chat",
                        "messages": [
                            {"role": "system", "content": SYSTEM_PROMPT},
                            {"role": "user", "content": user_prompt}
                        ],
                        "temperature": 0.1,  # 低温度，确保输出稳定
                        "response_format": {"type": "json_object"}  # 强制JSON输出
                    }
                )
                response.raise_for_status()
                
            # 3. 解析响应
            result = response.json()
            content = result["choices"][0]["message"]["content"]
            
            # 4. 提取JSON
            algorithm_params = json.loads(content)
            
            return {
                "success": True,
                "algorithm_params": algorithm_params,
                "raw_response": content
            }
            
        except httpx.TimeoutException:
            return {
                "success": False,
                "error": "DeepSeek API调用超时（30秒）",
                "algorithm_params": load_default_params()
            }
        except json.JSONDecodeError:
            return {
                "success": False,
                "error": "DeepSeek返回格式错误，无法解析JSON",
                "algorithm_params": load_default_params()
            }
        except Exception as e:
            return {
                "success": False,
                "error": f"DeepSeek API调用失败：{str(e)}",
                "algorithm_params": load_default_params()
            }
```

# 文件：src/backend/services/log_service.py

class LogService:
    """埋点日志服务"""
    
    @staticmethod
    def log_event(event_name: str, user_id: int, role: str, **kwargs):
        """
        记录埋点事件
        
        Args:
            event_name: 事件名称（login/logout/global_schedule/node_dispatch/route_plan/replan/deepseek_call）
            user_id: 用户ID
            role: 用户角色（dispatcher/manager）
            **kwargs: 事件特定数据
        """
        event_data_map = {
            "login": {"ip": kwargs.get("ip"), "user_agent": kwargs.get("user_agent")},
            "logout": {},
            "global_schedule": {
                "schedule_code": kwargs.get("schedule_code"),
                "order_count": kwargs.get("order_count"),
                "algorithm_type": kwargs.get("algorithm_type", "traditional")
            },
            "node_dispatch": {
                "batch_code": kwargs.get("batch_code"),
                "package_count": kwargs.get("package_count"),
                "vehicle_count": kwargs.get("vehicle_count"),
                "algorithm_type": kwargs.get("algorithm_type", "traditional")
            },
            "route_plan": {
                "route_count": kwargs.get("route_count"),
                "vehicle_count": kwargs.get("vehicle_count")
            },
            "replan": {
                "event_code": kwargs.get("event_code"),
                "reason": kwargs.get("reason"),
                "new_schedule_code": kwargs.get("new_schedule_code")
            },
            "deepseek_call": {
                "function_name": kwargs.get("function_name"),  # parse/explain/review/analyze
                "success": kwargs.get("success"),
                "degraded": kwargs.get("degraded", False)
            }
        }
        
        event_data = event_data_map.get(event_name, {})
        
        # 写入数据库
        from ...models.log_event import LogEvent
        from ...config.database import SessionLocal
        
        db = SessionLocal()
        try:
            log_event = LogEvent(
                event_name=event_name,
                user_id=user_id,
                role=role,
                event_data=event_data
            )
            db.add(log_event)
            db.commit()
        finally:
            db.close()
```

# 文件：src/backend/services/log_decorator.py

from functools import wraps
from ...services.log_service import LogService

def auto_log(event_name: str):
    """自动埋点装饰器"""
    def decorator(func):
        @wraps(func)
        async def wrapper(*args, **kwargs):
            # 执行原方法
            result = await func(*args, **kwargs)
            
            # 记录埋点（从kwargs中提取用户信息）
            user_id = kwargs.get("current_user_id")
            role = kwargs.get("current_user_role")
            
            if user_id and role:
                LogService.log_event(event_name, user_id, role, **result)
                
            return result
        return wrapper
    return decorator

# 使用示例

### 3.3 实现逻辑

```python
# 文件：src/backend/api/ai.py

@router.post("/ai/parse", response_model=AiParseResponse)
async def parse_natural_language(request: AiParseRequest, current_user: User = Depends(get_current_user)):
    """
    自然语言解析 → 调度执行
    
    流程：
    1. 调用DeepSeek API解析自然语言
    2. 如果DeepSeek失败，降级使用默认参数
    3. 如果auto_execute=true，自动执行完整调度链路（参考阶段4的demo_mode=true）
    """
    # 1. 获取系统上下文
    system_context = await get_system_context()
    
    # 2. 调用DeepSeek
    deepseek_result = await DeepSeekService.parse_natural_language(request.message, system_context)
    
    algorithm_params = deepseek_result["algorithm_params"]
    degraded = not deepseek_result["success"]
    degraded_reason = deepseek_result.get("error")
    
    # 3. 记录埋点
    LogService.log_event(
        "deepseek_call",
        current_user.id,
        current_user.role,
        function_name="parse",
        success=deepseek_result["success"],
        degraded=degraded
    )
    
    # 4. 如果auto_execute=true，执行完整调度链路
    schedule_code = None
    if request.auto_execute:
        # 参考阶段4的demo_mode=true逻辑
        # 4.1 执行F007全局调度
        schedule_result = await ScheduleService.create_global_schedule(
            current_user_id=current_user.id,
            current_user_role=current_user.role,
            algorithm_params=algorithm_params["global_schedule"]
        )
        schedule_code = schedule_result["schedule_code"]
        
        # 4.2 执行F005节点间调度（demo_mode=true，连续执行两次）
        batch_result = await DispatchService.create_node_dispatch(
            schedule_code=schedule_code,
            demo_mode=True,  # 关键：跳过模拟送达，连续执行L0→L1和L1→L2
            algorithm_params=algorithm_params["node_dispatch"]
        )
        batch_code = batch_result["batch_code"]
        
        # 4.3 执行F006路径规划
        await RouteService.create_routes(
            batch_code=batch_code,
            algorithm_params=algorithm_params["route_planning"]
        )
    
    return {
        "schedule_code": schedule_code,
        "algorithm_params": algorithm_params,
        "degraded": degraded,
        "degraded_reason": degraded_reason
    }
```
